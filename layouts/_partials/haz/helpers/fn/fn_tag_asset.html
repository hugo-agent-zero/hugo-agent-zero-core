{{/* fn_tag_asset.html Emit one stylesheet or script tag from an enqueue row. Args: - src (string, required): asset path, http URL, or static-… - type (string): style | script (default style; legacy: kind) - bust (bool): fingerprint pipeline assets (default true) - ver (string): query fragment e.g. "ver=XYZ" — stacked after bust (?/$ver or &/$ver) - context (any): ExecuteAsTemplate data (optional) */}} {{- $src := index . "src" | default "" -}} {{- $type := index . "type" | default (index . "kind" | default "style") -}} {{- $bust := true -}} {{- if isset . "bust" -}} {{- $bust = index . "bust" -}} {{- end -}} {{- /* YAML may stringify booleans; with false is empty in Go templates */ -}} {{- if eq (printf "%v" $bust) "false" -}} {{- $bust = false -}} {{- else if eq (printf "%v" $bust) "true" -}} {{- $bust = true -}} {{- end -}} {{- $ver := index . "ver" | default "" -}} {{- $tplCtx := index . "context" | default . -}} {{- if $src -}} {{- $href := "" -}} {{- if hasPrefix $src "http" -}} {{- $href = $src -}} {{- else if hasPrefix $src "static-" -}} {{- $href = strings.TrimPrefix "static-" $src | relURL -}} {{- else -}} {{- $res := resources.Get $src -}} {{- if $res -}} {{- if in $res.Content "{{" -}} {{- $res = $res | resources.ExecuteAsTemplate (printf "executed/%s" $src) $tplCtx -}} {{- end -}} {{- if $bust -}} {{- $res = $res | resources.Fingerprint -}} {{- end -}} {{- $href = $res.RelPermalink -}} {{- else -}} {{- warnf "enqueue asset not found: %s" $src -}} {{- end -}} {{- end -}} {{- if and $href (ne $ver "") -}} {{- if in $href "?" -}} {{- $href = printf "%s&%s" $href $ver -}} {{- else -}} {{- $href = printf "%s?%s" $href $ver -}} {{- end -}} {{- end -}} {{- if $href -}} {{- if eq $type "script" -}} {{- else -}} {{- end -}} {{- end -}} {{- end -}}
