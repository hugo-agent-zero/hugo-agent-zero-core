{{/* fn_shortcode_resolve_route_key.html Resolve route → shortcode profile → key_layouts_entry. Missing profile continues fallback; present enabled != true is a hard stop. Args: shortcodes, key_profile, context */}} {{- $shortcodes := index . "shortcodes" | default dict -}} {{- $keyProfile := index . "key_profile" | default "" -}} {{- $context := index . "context" -}} {{- $routeKey := index $shortcodes "route_key" | default dict -}} {{- $rules := index $routeKey "rules" | default slice -}} {{- $candidates := partial "haz/helpers/fn/fn_shortcode_route_candidates.html" (dict "context" $context) -}} {{- $selected := false -}} {{- $result := dict -}} {{- range $candidate := $candidates -}} {{- if not $selected -}} {{- range $rule := $rules -}} {{- if and (not $selected) (eq (index $rule "match" | default "") $candidate) (eq (index $rule "enabled") true) -}} {{- $keys := index $rule "keys" | default dict -}} {{- if isset $keys $keyProfile -}} {{- $selected = true -}} {{- $mapping := index $keys $keyProfile | default dict -}} {{- if ne (index $mapping "enabled") true -}} {{- $result = dict "__hard_stop" true "__reason" "disabled_route_key" -}} {{- else -}} {{- $entryKey := index $mapping "key_layouts_entry" | default "" -}} {{- if eq $entryKey "" -}} {{- $result = dict "__hard_stop" true "__reason" "missing_route_key_layouts_entry" -}} {{- else -}} {{- $result = dict "key_layouts_entry" $entryKey -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $selected -}} {{- $defaults := index $routeKey "defaults" | default dict -}} {{- $keys := index $defaults "keys" | default dict -}} {{- if not (isset $keys $keyProfile) -}} {{- $result = dict "__hard_stop" true "__reason" "missing_route_key_default" -}} {{- else -}} {{- $mapping := index $keys $keyProfile | default dict -}} {{- if ne (index $mapping "enabled") true -}} {{- $result = dict "__hard_stop" true "__reason" "disabled_route_key_default" -}} {{- else -}} {{- $entryKey := index $mapping "key_layouts_entry" | default "" -}} {{- if eq $entryKey "" -}} {{- $result = dict "__hard_stop" true "__reason" "missing_route_key_default_layouts_entry" -}} {{- else -}} {{- $result = dict "key_layouts_entry" $entryKey -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $result -}}
