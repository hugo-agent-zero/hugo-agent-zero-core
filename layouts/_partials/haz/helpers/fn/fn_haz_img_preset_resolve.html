{{/* fn_haz_img_preset_resolve.html Resolve presets[name] with enabled + enabled_false redirect. Flow per hop: 1. Look up key in presets (empty / missing / cycle → fail). 2. Enabled? omit or anything except false = on → use this preset. 3. Off → take enabled_false as the next key and hop again. No redirect key → fail (same as no match). Args: presets (dict), key (string) Return: dict key, preset, ok (bool) */}} {{- $presets := index . "presets" | default dict -}} {{- if not (reflect.IsMap $presets) -}} {{- $presets = dict -}} {{- end -}} {{- $key := strings.TrimSpace (printf "%v" (index . "key" | default "")) -}} {{- $visited := dict -}} {{- $preset := dict -}} {{- $ok := false -}} {{- range seq 1 32 -}} {{- if $ok -}} {{- else -}} {{- /* Fail gates before we touch enabled. */ -}} {{- if or (eq $key "") (isset $visited $key) (not (isset $presets $key)) -}} {{- $key = "" -}} {{- $preset = dict -}} {{- $ok = true -}} {{- else -}} {{- $visited = merge $visited (dict $key true) -}} {{- $cand := index $presets $key -}} {{- if not (reflect.IsMap $cand) -}} {{- $cand = dict -}} {{- end -}} {{- /* ASAP: omit → on (default true); only false-ish turns it off. */ -}} {{- $enabled := partial "haz/helpers/fn/fn_haz_bool.html" (dict "value" (index $cand "enabled") "default" true) -}} {{- if $enabled -}} {{- $preset = $cand -}} {{- $ok = true -}} {{- else -}} {{- $redir := "" -}} {{- if isset $cand "enabled_false" -}} {{- $redir = strings.TrimSpace (printf "%v" (index $cand "enabled_false")) -}} {{- end -}} {{- if eq $redir "" -}} {{- $key = "" -}} {{- $preset = dict -}} {{- $ok = true -}} {{- else -}} {{- $key = $redir -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $ok -}} {{- $key = "" -}} {{- $preset = dict -}} {{- end -}} {{- return (dict "key" $key "preset" $preset "ok" (and $ok (ne $key ""))) -}}
