{{/* fn_href_build.html Build final href from href only. Empty → fn_site_root_path. Already rooted there → leave it. Else trim a leading / (host-root) then relURL. */}} {{- $hrefBase := index . "href_base" | default "" -}} {{- $href := $hrefBase -}} {{- $root := partial "haz/helpers/fn/fn_site_root_path.html" -}} {{- if eq $href "" -}} {{- $href = $root -}} {{- else -}} {{- $isExplicit := partial "haz/helpers/fn/fn_href_is_explicit.html" (dict "href" $href) -}} {{- if not $isExplicit -}} {{- $rootTrim := strings.TrimSuffix $root "/" -}} {{- $alreadyRooted := and (ne $rootTrim "") (or (eq $href $root) (eq $href $rootTrim) (hasPrefix $href $root)) -}} {{- if not $alreadyRooted -}} {{- $href = partial "haz/helpers/fn/fn_href_trim_host_root.html" (dict "href" $href) | relURL -}} {{- end -}} {{- end -}} {{- end -}} {{- return $href -}}
