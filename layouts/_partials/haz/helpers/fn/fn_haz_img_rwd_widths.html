{{/* fn_haz_img_rwd_widths.html Return config.rwd_widths as key → { px, q: { jpeg } } (file menu for srcset w + process quality). Bare int → { px: N, q: { jpeg: -1 } }. Flat q: 75 → q.jpeg: 75. q.jpeg -1 or omitted = Hugo imaging default (no qN on Resize). No ladder / gutter math — stamp only. Args: - config (dict) settings.haz_img.config Swappable via config.helpers.rwd_widths_fn. */}} {{- $config := index . "config" | default dict -}} {{- $raw := index $config "rwd_widths" | default dict -}} {{- $out := dict -}} {{- range $k, $v := $raw -}} {{- $px := 0 -}} {{- $qJpeg := -1 -}} {{- if reflect.IsMap $v -}} {{- $px = int (index $v "px" | default 0) -}} {{- if isset $v "q" -}} {{- $qRaw := index $v "q" -}} {{- if reflect.IsMap $qRaw -}} {{- if isset $qRaw "jpeg" -}} {{- $qJpeg = int (index $qRaw "jpeg") -}} {{- end -}} {{- else -}} {{- $qJpeg = int $qRaw -}} {{- end -}} {{- end -}} {{- else -}} {{- $px = int $v -}} {{- end -}} {{- $out = merge $out (dict $k (dict "px" $px "q" (dict "jpeg" $qJpeg))) -}} {{- end -}} {{- /* OP dogfood fallbacks if rwd_widths missing. */ -}} {{- if eq (len $out) 0 -}} {{- $out = dict "xs" (dict "px" 240 "q" (dict "jpeg" -1)) "sm" (dict "px" 479 "q" (dict "jpeg" -1)) "md" (dict "px" 767 "q" (dict "jpeg" -1)) "lg" (dict "px" 1024 "q" (dict "jpeg" -1)) "xl" (dict "px" 1536 "q" (dict "jpeg" -1)) "xxl" (dict "px" 2048 "q" (dict "jpeg" -1)) -}} {{- end -}} {{- return $out -}}
