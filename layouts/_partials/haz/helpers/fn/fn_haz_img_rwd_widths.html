{{/* fn_haz_img_rwd_widths.html Return config.rwd_widths as key → { px, format, q: { jpeg, webp } }. Bare int → { px: N, format: inherit, q: { jpeg: -1, webp: -1 } }. format: inherit unless a valid value overrides it (this pass: inherit|webp). Empty/null/unknown → inherit. q.* -1/omit = Hugo imaging default (no qN on Resize). No ladder / gutter math — stamp only. Args: - config (dict) settings.haz_img.config Swappable via config.helpers.rwd_widths_fn. */}} {{- $config := index . "config" | default dict -}} {{- $raw := index $config "rwd_widths" | default dict -}} {{- $formatOk := slice "inherit" "webp" -}} {{- $out := dict -}} {{- range $k, $v := $raw -}} {{- $px := 0 -}} {{- $format := "inherit" -}} {{- $qJpeg := -1 -}} {{- $qWebp := -1 -}} {{- if reflect.IsMap $v -}} {{- $px = int (index $v "px" | default 0) -}} {{- if isset $v "format" -}} {{- $cand := lower (strings.TrimSpace (printf "%v" (index $v "format"))) -}} {{- if and $cand (in $formatOk $cand) -}} {{- $format = $cand -}} {{- end -}} {{- end -}} {{- if isset $v "q" -}} {{- $qRaw := index $v "q" -}} {{- if reflect.IsMap $qRaw -}} {{- if isset $qRaw "jpeg" -}} {{- $qJpeg = int (index $qRaw "jpeg") -}} {{- end -}} {{- if isset $qRaw "webp" -}} {{- $qWebp = int (index $qRaw "webp") -}} {{- end -}} {{- else -}} {{- $qJpeg = int $qRaw -}} {{- end -}} {{- end -}} {{- else -}} {{- $px = int $v -}} {{- end -}} {{- $out = merge $out (dict $k (dict "px" $px "format" $format "q" (dict "jpeg" $qJpeg "webp" $qWebp))) -}} {{- end -}} {{- /* OP dogfood fallbacks if rwd_widths missing. */ -}} {{- if eq (len $out) 0 -}} {{- $out = dict "xs" (dict "px" 240 "format" "inherit" "q" (dict "jpeg" -1 "webp" -1)) "sm" (dict "px" 479 "format" "inherit" "q" (dict "jpeg" -1 "webp" -1)) "md" (dict "px" 767 "format" "inherit" "q" (dict "jpeg" -1 "webp" -1)) "lg" (dict "px" 1024 "format" "inherit" "q" (dict "jpeg" -1 "webp" -1)) "xl" (dict "px" 1536 "format" "inherit" "q" (dict "jpeg" -1 "webp" -1)) "xxl" (dict "px" 2048 "format" "inherit" "q" (dict "jpeg" -1 "webp" -1)) -}} {{- end -}} {{- return $out -}}
