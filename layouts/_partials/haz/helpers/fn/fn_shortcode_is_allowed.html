{{/* fn_shortcode_is_allowed — settings.shortcodes.whitelist membership. Listed = allowed. Omit whitelist (or settings.shortcodes) → allow all. Off switch: _-park the list entry (lookup misses). Args: - context (page) - name (string) shortcode id, e.g. haz_img / haz_do - add (string, optional) haz_do verb; required for verb-filtered haz_do */}} {{- $name := strings.TrimSpace (printf "%v" (index . "name" | default "")) -}} {{- $add := lower (strings.TrimSpace (printf "%v" (index . "add" | default ""))) -}} {{- $ctx := index . "context" | default . -}} {{- $shortcodes := partial "haz/helpers/fn/fn_get_sys_manifest_prop.html" (dict "context" $ctx "path" (slice "settings" "shortcodes")) -}} {{- if not (reflect.IsMap $shortcodes) -}} {{- return true -}} {{- end -}} {{- if not (isset $shortcodes "whitelist") -}} {{- return true -}} {{- end -}} {{- $wl := index $shortcodes "whitelist" -}} {{- if not (reflect.IsSlice $wl) -}} {{- return true -}} {{- end -}} {{- $allowed := false -}} {{- range $item := $wl -}} {{- if $allowed -}} {{- else if reflect.IsMap $item -}} {{- if isset $item $name -}} {{- $verbs := index $item $name -}} {{- if reflect.IsSlice $verbs -}} {{- if eq $add "" -}} {{- $allowed = false -}} {{- else if in $verbs $add -}} {{- $allowed = true -}} {{- end -}} {{- else -}} {{- /* map entry with no verb list → SC allowed as a whole */ -}} {{- $allowed = true -}} {{- end -}} {{- end -}} {{- else -}} {{- if eq (printf "%v" $item) $name -}} {{- $allowed = true -}} {{- end -}} {{- end -}} {{- end -}} {{- return $allowed -}}
