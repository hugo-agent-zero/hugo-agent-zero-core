{{/* haz_do tab blade. Nested inside add="tabs". title is the label. Inner is Markdown unless html=true. html=true prints Inner as markup (safeHTML), same pass-through as haz_img. CSS radios switch panels. No script. */}} {{- $extra := "" -}} {{- with .Get "class" -}} {{- $extra = strings.TrimSpace . | htmlEscape -}} {{- end -}} {{- $group := .Ordinal -}} {{- with .Parent -}} {{- $group = .Ordinal -}} {{- end -}} {{- $key := printf "haz_do_tabs_%d" $group -}} {{- $n := .Page.Store.Get $key | default 0 -}} {{- .Page.Store.Set $key (add $n 1) -}} {{- $rid := printf "haz_do_tab_%d_%d" $group $n -}} {{- $rname := printf "haz_do_tabs_%d" $group -}} {{- $title := "Tab" -}} {{- with .Get "title" -}} {{- $t := strings.TrimSpace . -}} {{- if $t -}} {{- $title = $t -}} {{- end -}} {{- end -}} {{- $wantHTML := false -}} {{- if isset .Params "html" -}} {{- $htmlValue := .Get "html" -}} {{- if eq (printf "%T" $htmlValue) "bool" -}} {{- $wantHTML = $htmlValue -}} {{- else -}} {{- $htmlLow := lower (printf "%v" $htmlValue) -}} {{- $wantHTML = or (eq $htmlLow "true") (eq $htmlLow "1") (eq $htmlLow "yes") (eq $htmlLow "on") -}} {{- end -}} {{- end -}} {{- $raw := strings.TrimSpace .Inner -}} {{- $body := $raw | markdownify -}} {{- if $wantHTML -}} {{- $body = $raw | safeHTML -}} {{- end -}} {{- $labelClass := "haz_do_tab__label" -}} {{- $panelClass := "haz_do_tab__panel" -}} {{- if $extra -}} {{- $labelClass = printf "%s %s" $labelClass $extra -}} {{- $panelClass = printf "%s %s" $panelClass $extra -}} {{- end -}} {{- $panelKey := printf "haz_do_tabs_panels_%d" $group -}} {{- $panels := .Page.Store.Get $panelKey | default slice -}} {{- $panel := print (printf "
" $panelClass) $body "
" | safeHTML -}} {{- .Page.Store.Set $panelKey ($panels | append $panel) -}}
