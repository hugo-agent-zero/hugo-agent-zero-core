Any other type renders with the note icon and color until a .haz_do_callout--{type} rule sets the accent and icon. type=code does not emit pre or code tags by itself. Inner is Markdown. A fence in the body stays a fence, and the rest stays prose. No fence at all: the whole body is wrapped as one code block. */}} {{- $type := "note" -}} {{- with .Get "type" -}} {{- $cand := lower (strings.TrimSpace .) | replaceRE "[^a-z0-9_-]+" "" -}} {{- if $cand -}} {{- $type = $cand -}} {{- end -}} {{- end -}} {{- $class := printf "haz_do haz_do_callout haz_do_callout--%s" $type -}} {{- with .Get "class" -}} {{- $extra := strings.TrimSpace . -}} {{- if $extra -}} {{- $class = printf "%s %s" $class $extra -}} {{- end -}} {{- end -}} {{- $title := "" -}} {{- with .Get "title" -}} {{- $title = strings.TrimSpace . -}} {{- end -}} {{- $raw := strings.TrimSpace .Inner -}} {{- if and (eq $type "code") $raw (not (in $raw "```")) -}} {{- $raw = printf "```\n%s\n```" $raw -}} {{- end -}} {{- $body := $raw | markdownify -}} {{- $data := dict "haz_callout" $type -}} {{- with .Get "step" -}} {{- $step := strings.TrimSpace . -}} {{- if $step -}} {{- $data = merge $data (dict "haz_callout_step" $step) -}} {{- end -}} {{- end -}} {{- $rootAttrs := dict "class" $class "data" $data -}} {{- with .Get "id" -}} {{- $id := strings.TrimSpace . -}} {{- if $id -}} {{- $rootAttrs = merge $rootAttrs (dict "id" $id) -}} {{- end -}} {{- end -}} {{- partial "haz/helpers/fn/fn_tag_tag_open.html" (dict "tag" "div" "attrs" $rootAttrs) -}} {{- partial "haz/helpers/fn/fn_tag_tag_open.html" (dict "tag" "div" "attrs" (dict "class" "haz_do_callout__copy")) -}} {{- with $title }} {{- partial "haz/helpers/fn/fn_tag_tag_open.html" (dict "tag" "p" "attrs" (dict "class" "haz_do_callout__title")) -}} {{- . -}} {{- print "
