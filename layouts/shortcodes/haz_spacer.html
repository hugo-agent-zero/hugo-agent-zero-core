{{/* haz_spacer — air between blocks. Clear rides along. Usage: {{< haz_spacer >}} {{< haz_spacer size="2" >}} {{< haz_spacer size="0.5" clear=false >}} size rem, default 1. Floats ok. 0 is a fence with no air. clear default true. false skips the clear class. */}} {{- $size := 1.0 -}} {{- with .Get "size" -}} {{- $size = float . -}} {{- end -}} {{- $wantClear := true -}} {{- if isset .Params "clear" -}} {{- $clearValue := .Get "clear" -}} {{- if eq (printf "%T" $clearValue) "bool" -}} {{- $wantClear = $clearValue -}} {{- else -}} {{- $clearLow := lower (printf "%v" $clearValue) -}} {{- $wantClear = or (eq $clearLow "true") (eq $clearLow "1") (eq $clearLow "yes") (eq $clearLow "on") -}} {{- end -}} {{- end -}} {{- $class := "haz_spacer" -}} {{- if $wantClear -}} {{- $class = printf "%s haz_clear" $class -}} {{- end -}}
