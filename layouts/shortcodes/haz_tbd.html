{{/* haz_tbd — catch-all for a handful of stream helpers. Name is a placeholder. Usage: {{< haz_tbd type="spacer" >}} {{< haz_tbd type="spacer" size="2" >}} {{< haz_tbd type="spacer" size="0.5" clear=false >}} type required. spacer for now. size spacer: rem, default 1. Floats ok. 0 is a fence with no air. clear spacer: default true. false skips the clear class. */}} {{- $type := "" -}} {{- with .Get "type" -}} {{- $type = lower (strings.TrimSpace .) -}} {{- end -}} {{- if eq $type "spacer" -}} {{- $size := 1.0 -}} {{- with .Get "size" -}} {{- $size = float . -}} {{- end -}} {{- $wantClear := true -}} {{- if isset .Params "clear" -}} {{- $clearValue := .Get "clear" -}} {{- if eq (printf "%T" $clearValue) "bool" -}} {{- $wantClear = $clearValue -}} {{- else -}} {{- $clearLow := lower (printf "%v" $clearValue) -}} {{- $wantClear = or (eq $clearLow "true") (eq $clearLow "1") (eq $clearLow "yes") (eq $clearLow "on") -}} {{- end -}} {{- end -}} {{- $class := "haz_spacer" -}} {{- if $wantClear -}} {{- $class = printf "%s haz_clear" $class -}} {{- end -}}
{{- else -}} {{- end -}}
