{{/* haz_do — stream toolbox dispatcher. Not WordPress. Usage (MD → markup): {{< haz_do add="spacer" >}} →
{{< haz_do add="spacer" height="2" >}} →
{{< haz_do add="spacer" height="24" height_unit="px" >}} →
{{< haz_do add="spacer" clear=false >}} →
{{< haz_do add="spacer" class="haz_my_gap" >}} →
add verb. Allowlist: spacer. height spacer: number. Omitted → CSS default (.haz_do_spacer). height_unit spacer: CSS length unit. Default rem. Allowlist: rem em px vh vw %. clear spacer: default true. false skips haz_clear (float utility). class appended after kit classes. Escaped. Does not replace them. */}} {{- $add := "" -}} {{- with .Get "add" -}} {{- $add = lower (strings.TrimSpace .) -}} {{- end -}} {{- $allowedAdds := slice "spacer" -}} {{- if not (in $allowedAdds $add) -}} {{- else if eq $add "spacer" -}} {{- $class := "haz_do haz_do_spacer" -}} {{- $wantClear := true -}} {{- if isset .Params "clear" -}} {{- $clearValue := .Get "clear" -}} {{- if eq (printf "%T" $clearValue) "bool" -}} {{- $wantClear = $clearValue -}} {{- else -}} {{- $clearLow := lower (printf "%v" $clearValue) -}} {{- $wantClear = or (eq $clearLow "true") (eq $clearLow "1") (eq $clearLow "yes") (eq $clearLow "on") -}} {{- end -}} {{- end -}} {{- if $wantClear -}} {{- $class = printf "%s haz_clear" $class -}} {{- end -}} {{- with .Get "class" -}} {{- $extra := strings.TrimSpace . | htmlEscape -}} {{- if $extra -}} {{- $class = printf "%s %s" $class $extra -}} {{- end -}} {{- end -}} {{- $args := dict "class" $class -}} {{- if isset .Params "height" -}} {{- $unit := "rem" -}} {{- $allowedUnits := slice "rem" "em" "px" "vh" "vw" "%" -}} {{- with .Get "height_unit" -}} {{- $u := lower (strings.TrimSpace .) -}} {{- if in $allowedUnits $u -}} {{- $unit = $u -}} {{- end -}} {{- end -}} {{- $args = merge $args (dict "height" (printf "%v%s" (float (.Get "height")) $unit)) -}} {{- end -}} {{- partial "haz/helpers/fn/fn_tag_div_spacer.html" $args -}} {{- end -}}
